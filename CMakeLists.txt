cmake_minimum_required(VERSION 3.21)
project(GaomonManagePanel VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src")
include(GNUInstallDirs)
configure_file(src/startup-paths.h.in generated/startup-paths.h @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
qt_add_executable(gaomon-manage-panel
  src/main.cpp
  src/area-preview.h
  src/pressure-curve.h
)
target_include_directories(gaomon-manage-panel PRIVATE "${SRC_DIR}")
qt_add_resources(gaomon-manage-panel app_icons PREFIX "/" FILES assets/gaomon-manage-panel.png)
set(SHORTCUT_PREVIEW_IMAGES assets/shortcut-preview/tablet.png assets/shortcut-preview/pen.png)
qt_add_resources(gaomon-manage-panel shortcut_images PREFIX "/" FILES ${SHORTCUT_PREVIEW_IMAGES})
target_link_libraries(gaomon-manage-panel PRIVATE Qt6::Widgets Qt6::DBus)
install(TARGETS gaomon-manage-panel RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES resources/gaomon-manage-panel.desktop DESTINATION share/applications)
install(FILES LICENSE DESTINATION share/licenses/gaomon-manage-panel)
install(FILES assets/NOTICE.md DESTINATION share/doc/gaomon-manage-panel)
install(FILES assets/gaomon-manage-panel.png DESTINATION share/icons/hicolor/128x128/apps)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBEVDEV REQUIRED IMPORTED_TARGET libevdev)
qt_add_executable(gaomon-shortcut-helper src/shortcut-helper.cpp)
target_include_directories(gaomon-shortcut-helper PRIVATE "${SRC_DIR}")
target_link_libraries(gaomon-shortcut-helper PRIVATE Qt6::Core PkgConfig::LIBEVDEV)
install(TARGETS gaomon-shortcut-helper RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
qt_add_executable(gaomon-permission-setup src/permission-setup.cpp)
target_link_libraries(gaomon-permission-setup PRIVATE Qt6::Core)
install(TARGETS gaomon-permission-setup RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
include(CTest)
