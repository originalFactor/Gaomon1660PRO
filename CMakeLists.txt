cmake_minimum_required(VERSION 3.21)
project(GaomonManagePanel VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
qt_add_executable(gaomon-manage-panel main.cpp area-preview.h pressure-curve.h)
qt_add_resources(gaomon-manage-panel app_icons PREFIX "/" FILES assets/gaomon-manage-panel.png)
target_link_libraries(gaomon-manage-panel PRIVATE Qt6::Widgets Qt6::DBus)
install(TARGETS gaomon-manage-panel RUNTIME DESTINATION bin)
install(FILES gaomon-manage-panel.desktop DESTINATION share/applications)
install(FILES LICENSE DESTINATION share/licenses/gaomon-manage-panel)
install(FILES assets/NOTICE.md DESTINATION share/doc/gaomon-manage-panel)
install(FILES assets/gaomon-manage-panel.png DESTINATION share/icons/hicolor/128x128/apps)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBEVDEV REQUIRED IMPORTED_TARGET libevdev)
qt_add_executable(gaomon-shortcut-helper shortcut-helper.cpp)
target_link_libraries(gaomon-shortcut-helper PRIVATE Qt6::Core PkgConfig::LIBEVDEV)
install(TARGETS gaomon-shortcut-helper RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(area-preview-test tests/area-preview-test.cpp area-preview.h)
  target_link_libraries(area-preview-test PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME area-preview COMMAND area-preview-test)
  set_tests_properties(area-preview PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  qt_add_executable(pressure-sync-test tests/pressure-sync-test.cpp)
  target_link_libraries(pressure-sync-test PRIVATE Qt6::Core Qt6::DBus Qt6::Test)
  add_test(NAME pressure-sync COMMAND dbus-run-session -- ${CMAKE_COMMAND} -E env QT_QPA_PLATFORM=offscreen GAOMON_TEST_APP=$<TARGET_FILE:gaomon-manage-panel> $<TARGET_FILE:pressure-sync-test>)
  qt_add_executable(pressure-curve-test tests/pressure-curve-test.cpp pressure-curve.h)
  target_link_libraries(pressure-curve-test PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME pressure-curve COMMAND pressure-curve-test)
  set_tests_properties(pressure-curve PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(action-state-test tests/action-state-test.cpp)
  target_link_libraries(action-state-test PRIVATE Qt6::Core)
  add_test(NAME action-state COMMAND action-state-test)
  add_executable(shortcut-binding-test tests/shortcut-binding-test.cpp)
  target_link_libraries(shortcut-binding-test PRIVATE Qt6::Core)
  add_test(NAME shortcut-binding COMMAND shortcut-binding-test)
  add_executable(pen-buttons-test tests/pen-buttons-test.cpp)
  add_test(NAME pen-buttons COMMAND pen-buttons-test)
  add_executable(keyboard-matcher-test tests/keyboard-matcher-test.cpp)
  add_test(NAME keyboard-matcher COMMAND keyboard-matcher-test)
endif()
